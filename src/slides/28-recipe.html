function randomRamp(count, maxSpacing = 60) {
  const spacing = min(maxSpacing, 360 / count), start = random(360);
  const ring = Array.from({ length: round(360 / spacing) }, (_, i) => (start + i * spacing) % 360);
  const hues = shuffle(ring).slice(0, count).sort((a, b) => a - b);
  const minL = 0.05 + random(0.2),   maxL = 1 - random(0.15), rangeL = max(maxL - minL, 0.1);
  const minC = 0.005 + random(0.25), maxC = 1 - random(),     rangeC = max(maxC - minC, 0.1);
  const flip = random() < 0.5;
  return hues.map((h, i) => { const t = i / (count - 1);
    return chroma.oklch(minL + t * rangeL, 0.4 * (minC + (flip ? 1 - t : t) * rangeC), h); });
}

function scaleSpreadArray(initial, n, fill = (t, a, b) => chroma.mix(a, b, t, 'oklab')) {
  const chunks = initial.map((v) => [v]);
  for (let i = 0; i < n - initial.length; i++) chunks[i % (initial.length - 1)].push(null);
  for (let i = 0; i < chunks.length - 1; i++)
    for (let j = 1; j < chunks[i].length; j++) chunks[i][j] = fill(j / chunks[i].length, chunks[i][0], chunks[i + 1][0]);
  return chunks.flat();
}
