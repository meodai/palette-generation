function randomRamp(count, maxSpacing = 60) {
  const spacing = Math.min(maxSpacing, 360 / count);
  const startHue = rnd() * 360;

  // Evenly spaced candidate hues with a random rotation.
  const hues = Array.from(
    { length: Math.round(360 / spacing) },
    (_, i) => (startHue + i * spacing) % 360
  );

  // Randomly select the desired number of hues while preserving their order.
  const selectedHues = shuffle(hues)
    .slice(0, count)
    .sort((a, b) => a - b);

  // Random lightness and chroma ranges.
  const minLightness = 0.05 + rnd() * 0.20;
  const maxLightness = 1 - rnd() * 0.15;

  const minChroma = 0.005 + rnd() * 0.25;
  const maxChroma = 1 - rnd();

  const lightnessRange = Math.max(maxLightness - minLightness, 0.1);
  const chromaRange = Math.max(maxChroma - minChroma, 0.1);

  const reverseChroma = rnd() < 0.5;

  return selectedHues.map((hue, index) => {
    const t = index / (count - 1);

    return {
      h: hue,
      li: minLightness + t * lightnessRange,
      ch: minChroma + (reverseChroma ? 1 - t : t) * chromaRange,
    };
  });
}
