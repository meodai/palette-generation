All of it, in one function

function randomRamp(count, maxSpacing = 60) {
  const spacing = Math.min(maxSpacing, 360 / count);
  const start = rnd() * 360;

  const ring = Array.from({ length: Math.round(360 / spacing) },
    (_, i) => (start + i * spacing) % 360);

  const hues = shuffle(ring).slice(0, count).sort((a, b) => a - b);

  const minL = 0.05 + rnd() * 0.2,   maxL = 1 - rnd() * 0.15;
  const minC = 0.005 + rnd() * 0.25, maxC = 1 - rnd();
  const rangeL = Math.max(maxL - minL, 0.1);
  const rangeC = Math.max(maxC - minC, 0.1);
  const flip = rnd() < 0.5;

  return hues.map((h, i) => {
    const t = i / (count - 1);
    return { h, li: minL + t * rangeL, ch: minC + (flip ? 1 - t : t) * rangeC };
  });
}

The marked line is the one that matters: a random subset of the ring, put back in wheel order. Gaps round the wheel, not one arc. Everything after it is two linear walks and a coin flip.
