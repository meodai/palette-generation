function palette(n, stops = 3, spacing = 60) {
  const start = random(360);
  const ring = Array.from({ length: 360 / spacing }, (_, i) => (start + i * spacing) % 360);
  const hues = shuffle(ring).slice(0, stops).sort((a, b) => a - b);

  const anchors = hues.map((h, i) => {
    const t = i / (stops - 1);
    return `oklch(${lerp(15, 90, t)}% ${lerp(0.05, 0.18, t)} ${h})`;
  });

  return scaleSpreadArray(anchors, n);
}
